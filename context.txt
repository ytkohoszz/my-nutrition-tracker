## File & folder conventions

function findNutritionSectionBounds(lines: string[]): {
  headingIdx: number;
  endIdx: number;
} | null {
  let headingIdx = -1;
  for (let i = 0; i < lines.length; i++) {
    const trimmed = (lines[i] ?? "").trim();
    if (/^##\s+Питание\s*$/i.test(trimmed)) {
      headingIdx = i;
      break;
    }
  }
  if (headingIdx === -1) return null;

  let endIdx = lines.length;
  for (let i = headingIdx + 1; i < lines.length; i++) {
    const trimmed = (lines[i] ?? "").trim();
    if (/^#{1,2}\s+\S/.test(trimmed)) {
      endIdx = i;
      break;
    }
  }
  return { headingIdx, endIdx };
}

export function parseMealsFromDiaryContent(content: string): LoggedMealEntry[] {
  const lines = content.split(/\r?\n/);
  const bounds = findNutritionSectionBounds(lines);
  if (!bounds) return [];

  const entries: LoggedMealEntry[] = [];
  for (let i = bounds.headingIdx + 1; i < bounds.endIdx; i++) {
    const line = (lines[i] ?? "").trim();
    if (!line.startsWith("-")) continue;
    const m = line.match(MEAL_LINE_REGEX);
    if (!m) continue;

    const timeStr = (m[1] ?? "").trim();
    const name = (m[2] ?? "").trim();
    const grams = Number((m[3] ?? "0").replace(",", "."));
    const kcal = Number((m[4] ?? "0").replace(",", "."));
    const protein = Number((m[5] ?? "0").replace(",", "."));
    const fat = Number((m[6] ?? "0").replace(",", "."));
    const carbs = Number((m[7] ?? "0").replace(",", "."));

    if (!name || !Number.isFinite(grams)) continue;

    entries.push({
      lineIndex: i,
      timeStr,
      name,
      grams: round1(grams),
      kcal: Math.round(kcal),
      protein: round1(protein),
      fat: round1(fat),
      carbs: round1(carbs),
    });
  }
  return entries;
}

export function sumDailyMacros(entries: LoggedMealEntry[]): DailyMacrosSummary {
  let kcal = 0;
  let protein = 0;
  let fat = 0;
  let carbs = 0;
  for (const e of entries) {
    kcal += e.kcal;
    protein += e.protein;
    fat += e.fat;
    carbs += e.carbs;
  }
  return {
    kcal: Math.round(kcal),
    protein: round1(protein),
    fat: round1(fat),
    carbs: round1(carbs),
  };
}

function sanitizeFoodItem(raw: unknown): FoodItem | null {
  if (!raw || typeof raw !== "object") return null;
  const f = raw as Record<string, unknown>;
  const name = typeof f.name === "string" ? f.name.trim() : "";
  if (!name) return null;

  const num = (v: unknown, def = 0): number => {
    const n = typeof v === "number" ? v : Number(String(v ?? "").replace(",", "."));
    return Number.isFinite(n) && n >= 0 ? round1(n) : def;
  };

  const id =
    typeof f.id === "string" && f.id.trim() ? f.id.trim() : makeFoodId();

  return {
    id,
    name,
    kcalPer100: num(f.kcalPer100, 0),
    proteinPer100: num(f.proteinPer100, 0),
    fatPer100: num(f.fatPer100, 0),
    carbsPer100: num(f.carbsPer100, 0),
    defaultGrams: num(f.defaultGrams, 100) || 100,
    isFavorite: Boolean(f.isFavorite),
    useCount: Math.max(0, Math.round(num(f.useCount, 0))),

            .filter((x): x is FoodItem => x !== null);

    const filtered = q
      ? this.foods.filter((f) => f.name.toLowerCase().includes(q))
      : [...this.foods];

    return filtered.sort((a, b) => {
      if (Boolean(a.isFavorite) !== Boolean(b.isFavorite)) {
        return a.isFavorite ? -1 : 1;
      }
      if (b.lastUsedAt !== a.lastUsedAt) {
        return b.lastUsedAt - a.lastUsedAt;
      }
      if (b.useCount !== a.useCount) {
        return b.useCount - a.useCount;
      }
      return a.name.localeCompare(b.name, "ru");
    });

    return this.foods.find((f) => f.id === id);

      existing.isFavorite = Boolean(draft.isFavorite);

      return existing;
    }

      fatPer100: round1(draft.fatPer100),
      carbsPer100: round1(draft.carbsPer100),
      defaultGrams: round1(draft.defaultGrams || 100),
      isFavorite: Boolean(draft.isFavorite),
      useCount: 1,
      lastUsedAt: Date.now(),
    };

  recordFoodUsage(foodId: string, usedGrams: number): void {
    const item = this.findById(foodId);
    if (!item) return;
    item.useCount = (item.useCount || 0) + 1;
    item.lastUsedAt = Date.now();
    if (usedGrams > 0) {
      item.defaultGrams = round1(usedGrams);
    }
    this.requestSave();
  }

  toggleFavorite(foodId: string): void {
    const item = this.findById(foodId);
    if (!item) return;
    item.isFavorite = !item.isFavorite;
    this.requestSave();
  }

  deleteFood(foodId: string): boolean {
    const idx = this.foods.findIndex((f) => f.id === foodId);
    if (idx === -1) return false;
    this.foods.splice(idx, 1);
    this.requestSave();
    return true;
  }

    this.saveTimer = window.setTimeout(() => {
      this.saveTimer = null;
      void this.saveNow();

      const payload: FoodCatalogFileData = {
        version: 2,
        foods: this.foods,
      };

      const existing = this.app.vault.getAbstractFileByPath(this.dbPath);
      if (existing instanceof TFile) {
        await this.app.vault.modify(existing, jsonText);
      } else {
        await this.app.vault.create(this.dbPath, jsonText);
      }
    } catch {
      // игнорируем ошибки записи
    }
  }

  flush(): void {
    if (this.saveTimer !== null) {
      window.clearTimeout(this.saveTimer);
      this.saveTimer = null;
      void this.saveNow();
    }
  }

      if (submitBtn.disabled) return;
      const effGrams = round1(Number(gInp.value.replace(",", ".")) || 0);
      if (!Number.isFinite(effGrams) || effGrams <= 0) {
        new Notice("Введи количество граммов больше 0.");
        gInp.focus();
        return;

      const calc = this.calculatePortionMacros(activeFood, effGrams);

          grams: effGrams,
          ...calc,

          `Добавлено: ${activeFood.name} (${effGrams} г · ${calc.kcal} ккал)`

          "Ошибка записи питания: " +
            (err instanceof Error ? err.message : String(err))
        );
        submitBtn.disabled = false;
      }
    };

      if (e.key === "Enter") {
        e.preventDefault();
        void doSubmitMeal();
      }
    };
    submitBtn.onclick = () => void doSubmitMeal();

    updatePortionPreview();
  }

  private renderLoggedMeals(
    entries: LoggedMealEntry[],
    diaryFile: ReturnType<typeof findDiaryFileForDate>
  ): void {
    this.logEl.empty();

const head = this.logEl.createDiv("my-nt-section-head");
    head.createSpan({
      cls: "my-nt-section-title",
      text: entries.length > 0 ? `Съедено · ${entries.length}` : "Съедено",
    });

    if (entries.length === 0) {
      this.logEl.createDiv({
        cls: "my-nt-empty",
        text: diaryFile
          ? "В разделе «## Питание» за сегодня пока нет записей."
          : "Записей за сегодня пока нет.",
      });
      return;

    const listEl = this.logEl.createDiv("my-nt-meals-list");
    entries.forEach((e) => {

      const left = row.createDiv("my-nt-meal-left");
      if (e.timeStr) {
        left.createSpan({ cls: "my-nt-meal-time", text: e.timeStr });
      }
      left.createSpan({ cls: "my-nt-meal-name", text: e.name });
      left.createSpan({ cls: "my-nt-meal-amount", text: `${e.grams} г` });

      const right = row.createDiv("my-nt-meal-right");
      right.createSpan({
        cls: "my-nt-meal-kcal",
        text: `${e.kcal} ккал`,
      });
      right.createSpan({
        cls: "my-nt-meal-macros",
        text: `Б:${e.protein} Ж:${e.fat} У:${e.carbs}`,
      });

      const delBtn = right.createEl("button", {
        cls: "my-nt-meal-del",
        attr: { type: "button", title: "Удалить приём пищи" },
      });
      setIcon(delBtn, "x");

        delBtn.disabled = true;
        await deleteMealFromDiary(
          this.app,
          this.plugin.settings,
          this.dateISO,
          e.lineIndex
        );
        await this.refreshAll();
      };
    });
  }
